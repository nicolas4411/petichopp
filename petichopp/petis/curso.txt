  from django.db import models
  class Pet(models.Model):
    nome = models.CharField(max_length=100)
    especie = models.CharField(max_length=50)
    dono = models.CharField(max_length=100)
    ativo = models.BooleanField(default=True)  # Soft Delete: True (ativo) ou False (desativado)

    def __str__(self):
        return f"{self.nome} ({self.especie}) - Tutor: {self.dono}"
 
  exp no admin.py
    from django.contrib import admin
    from .models import Pet

    @admin.register(Pet)
    class PetAdmin(admin.ModelAdmin):
      # Colunas visíveis na tabela do painel
      list_display = ('id', 'nome', 'especie', 'dono', 'ativo')
    
      # Filtro lateral (permite filtrar ativos/inativos e por espécie)
      list_filter = ('ativo', 'especie')
    
      # Barra de pesquisa (busca por nome do pet ou do dono)
      search_fields = ('nome', 'dono')
